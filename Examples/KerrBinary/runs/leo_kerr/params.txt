# KerrBinary leo_kerr: hole A of runs/leo_flyby alone (Kerr M = 1, a/M = 0.95 along +z, at the
# origin), its exact slice everywhere (kerrbin.only_A: the same .kid, B and the CTT solution
# ignored), on the same grid, gauge and finders as leo_flyby. Tests the code on GPUs and
# measures A on its own: horizon size once the gauge settles, mass and spin drift at dx 1/64,
# junk radiation, and the speed to scale leo_flyby's run time from.
# Cells across A's horizon (diameter / dx): ~20 at t = 0 (r 0.156), ~35 by t ~ 6 (r 0.27).
#
# Leonardo: cd runs/leo_kerr && sbatch ../leonardo.slurm  (1 node, 4 A100), ~1-3 h.
# A first check: sbatch --qos=boost_qos_dbg --time=00:30:00 ../leonardo.slurm

amr.verbose = 1
amrex.use_gpu_aware_mpi = 1

#################################################
# Output (rank r's stdout: pout/pout.r)
# grteclyn.output_path = .

# amr.restart = checkpoints/chk00100   # continue from t = 100

# Number of level-0 time steps between outputs (coarse dt = 0.5)
amr.check_int = 40                 # every 20 M, ~5 GB each
amr.plot_int  = 8                  # every 4 M, ~3 GB each

amr.plot_vars        = chi lapse K Theta shift1 shift2
amr.derive_plot_vars = constraints Weyl4

#################################################
# Initial data (kerrbin)

kerrbin.kid_file = initial_data.kid   # relative to the run directory (the same file as leo_flyby)
kerrbin.only_A   = true               # A alone; the tracker's second puncture sits on A

#################################################
# Grid

amr.n_cell           = 256 256 128
geometry.prob_extent = 512 512 256
geometry.center      = 256 256 0   # A, the .kid origin, on the symmetry plane

# Outer boundary 256 from A: reflections reach the r = 100 extraction only after t ~ 290.
# Coarsest dx 2 costs little on GPUs (the finest levels are ~95% of the work).
amr.max_level = 7   # dx: 2, 1, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625

# Regrid intervals in level steps for levels 0..6: levels 2-6 every 0.125 M (B peaks at
# v ~ 0.55, moving 0.07 M = 4.4 finest cells between regrids, inside the 4-cell buffers).
amr.regrid_int = 4 4 1 2 4 8 16

tagging.threshold = 0.05

# Puncture tagging: finest level covers r < finest_level_factor * m around each hole, each
# coarser level 1.5x bigger: A r < 1.5, B r < 0.75 (B's horizon grows to r ~ 0.42)
puncture_tagging.finest_level_factor = 1.5

# GPU box sizes
amr.max_grid_size   = 64
amr.blocking_factor = 16
amr.n_error_buf     = 4

# skip rebuilding (and refilling from kerrbin) levels whose grids did not change
amr.use_efficient_regrid = 1

#################################################
# Boundaries

geometry.is_periodic  = 0 0 0
boundary.hi_condition = "SOMMERFELD_BC" "SOMMERFELD_BC" "SOMMERFELD_BC"
boundary.lo_condition = "SOMMERFELD_BC" "SOMMERFELD_BC" "REFLECTIVE_BC"   # z = 0 symmetry plane

#################################################
# Evolution

# evolution.dt_multiplier = 0.25
evolution.stop_time = 100.0    # gauge settled, junk radiation past all extraction radii
evolution.max_steps = -1

evolution.spatial_derivative_order = 4

# Moving-puncture gauge (as leo_flyby)
gauge.shift_advec_coeff = 1.0
# gauge.eta = 1.0   # ~ 1/M

evolution.sigma = 1.0          # Kreiss-Oliger dissipation

#################################################
# Weyl extraction, centred on geometry.center (A).
# all radii on level 1 (dx 1), as leo_flyby: A's junk radiation.

weyl_extraction.enabled          = true
weyl_extraction.num_radii        = 4
weyl_extraction.radii            = 40.0 60.0 80.0 100.0
weyl_extraction.levels           = 1    1    1    1
weyl_extraction.num_points_phi   = 24
weyl_extraction.num_points_theta = 37
weyl_extraction.num_modes = 12   # l = 2..4, m >= 0 (z -> -z symmetry: m < 0 follow)
weyl_extraction.modes     = 2 0  2 1  2 2  3 0  3 1  3 2  3 3  4 0  4 1  4 2  4 3  4 4

#################################################
# Puncture tracking

puncture_tracking.enabled = true
puncture_tracking.level   = 4      # dx = 0.125

#################################################
# SAHF horizon finders on every time step of sahf.level, each from its last solution,
# recentred on it; after a failure a finder restarts from the sphere f0 about its last
# centre. A restart starts each from its last row in sahf_output/horizon_<id>.dat.

sahf.n     = 1
sahf.level = 2                  # dt = 0.125
sahf.x     = 256.0              # A
sahf.y     = 256.0
sahf.z     = 0.0                # on the symmetry plane
sahf.f0    = 0.6                # initial and reset radius, outside the horizon
sahf.t0    = 0.0
sahf.tol      = 1e-2            # found when L2|Theta| < tol (if A is lost, raise it here and in leo_flyby)
sahf.max_iter = 60              # warm finds take 1-5 steps, a cold one ~20-30
sahf.min_r    = 1e-4
sahf.max_r    = 10.0
