# KerrBinary leo_kerr: exact Kerr, hole A of runs/leo_flyby alone (M = 1, a/M = 0.95 along
# +z, at the origin) with its exact slice everywhere (use_kerrbin = false: the same .kid,
# B and the CTT solution ignored), on the same grid, gauge and finders as leo_flyby. Tests
# the code on GPUs and measures A on its own: horizon size once the gauge settles, mass and
# spin drift at dx 1/64, junk radiation, and the speed to scale leo_flyby's run time from.
# Cells across A's horizon (diameter / dx): ~20 at t = 0 (r 0.156), ~35 by t ~ 6 (r 0.27).
#
# Leonardo: cd $CINECA_SCRATCH/KERR_PROJ/leo_kerr && sbatch jobscript.slurm (1 node,
# 4 A100), ~1-3 h. A first check: sbatch --qos=boost_qos_dbg --time=00:30:00 jobscript.slurm

amr.verbose = 1
amrex.use_gpu_aware_mpi = 1

#################################################
# Output in the run directory (plots, checkpoints, pout/pout.r, extraction_data,
# punctures_output, sahf_output). Restart: RESTART=chk00040 sbatch jobscript.slurm
# grteclyn.output_path = .

# Number of level-0 time steps between outputs (coarse dt = 1)
amr.check_int = 40                 # every 40 M, ~11 GB each
amr.plot_int  = 10                 # every 10 M, ~5.5 GB each

amr.plot_vars        = chi lapse K Theta shift1 shift2
amr.derive_plot_vars = constraints Weyl4

#################################################
# Initial data type: true = the kerrbin binary solution of kerrbin.kid_file;
# false = exact Kerr, hole A of that file alone (B's tracked puncture sits on A)

use_kerrbin      = false
kerrbin.kid_file = initial_data.kid   # the same file as leo_flyby: A's parameters

#################################################
# Grid

# Outer boundary 512 from A: echoes of the initial gauge pulse reach r = 160 only after
# t ~ 600. dx 4 only beyond r ~ 290, outside all extraction spheres.
amr.n_cell           = 256 256 128
geometry.prob_extent = 1024 1024 512
geometry.center      = 512 512 0   # A, the .kid origin, on the symmetry plane

amr.max_level = 8   # dx: 4, 2, 1, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625

# Regrid intervals in level steps for levels 0..7: levels 3-7 every 0.125 M (B peaks at
# v ~ 0.55, moving 0.07 M = 4.4 finest cells between regrids, inside the 4-cell buffers)
amr.regrid_int = 2 4 4 1 2 4 8 16

tagging.threshold = 0.05

# Puncture tagging: finest level covers r < finest_level_factor * m around each hole, each
# coarser level 1.5x bigger: A r < 1.5, B r < 0.75 (B's horizon grows to r ~ 0.42)
puncture_tagging.finest_level_factor = 1.5

# GPU box sizes
amr.max_grid_size   = 64
amr.blocking_factor = 16
amr.n_error_buf     = 4

# skip rebuilding (and refilling from kerrbin) levels whose grids did not change
amr.use_efficient_regrid = 1

#################################################
# Boundaries

geometry.is_periodic  = 0 0 0
boundary.hi_condition = "SOMMERFELD_BC" "SOMMERFELD_BC" "SOMMERFELD_BC"
boundary.lo_condition = "SOMMERFELD_BC" "SOMMERFELD_BC" "REFLECTIVE_BC"   # z = 0 symmetry plane

#################################################
# Evolution

# evolution.dt_multiplier = 0.25
evolution.stop_time = 100.0    # gauge settled; junk radiation past r = 40, 80
evolution.max_steps = -1

evolution.spatial_derivative_order = 4

# Moving-puncture gauge (as leo_flyby)
gauge.shift_advec_coeff = 1.0
# gauge.eta = 1.0   # ~ 1/M

evolution.sigma = 1.0          # Kreiss-Oliger dissipation

#################################################
# Weyl extraction, centred on geometry.center (A), all radii on level 2 (dx 1), as
# leo_flyby: A's junk radiation

weyl_extraction.enabled          = true
weyl_extraction.num_radii        = 4
weyl_extraction.radii            = 40.0 80.0 120.0 160.0
weyl_extraction.levels           = 2    2    2     2
weyl_extraction.num_points_phi   = 24
weyl_extraction.num_points_theta = 37
weyl_extraction.num_modes = 12   # l = 2..4, m >= 0 (z -> -z symmetry: m < 0 follow)
weyl_extraction.modes     = 2 0  2 1  2 2  3 0  3 1  3 2  3 3  4 0  4 1  4 2  4 3  4 4

#################################################
# Puncture tracking

puncture_tracking.enabled = true
puncture_tracking.level   = 5      # dx = 0.125

#################################################
# SAHF horizon finders on every time step of sahf.level, each from its last solution,
# recentred on it; after a failure a finder restarts from the sphere f0 about its last
# centre. A restart starts each from its last row in sahf_output/horizon_<id>.dat.

sahf.n     = 1
sahf.level = 3                  # dt = 0.125
sahf.x     = 512.0              # A
sahf.y     = 512.0
sahf.z     = 0.0                # on the symmetry plane
sahf.f0    = 0.6                # initial and reset radius, outside the horizon
sahf.t0    = 0.0
sahf.tol      = 2e-4            # found when rms Theta < tol (sqrt(int Theta^2 dA / area)); also
                                # measures the dx 1/64 floor (rmsTheta column)
sahf.max_iter = 60              # warm finds take 1-5 steps, a cold one ~20-30
sahf.min_r    = 1e-4
sahf.max_r    = 10.0
