# Kerr + boosted Schwarzschild fly-by (hyperbolic scattering), local CPU test
# REQUIRES a TwoPunctures build (spin is not supported by the default
# BinaryBHInitialData): make ... TWOPUNCTURES_SOURCE=/path/to/TwoPunctures/Source
#
# Physical set-up (units of total mass M = 1, each BH m = 0.5):
#   BH "plus"  = Kerr,          at rest,             chi = S/m^2 = 0.7 along +z
#   BH "minus" = Schwarzschild, v = 0.5 (P = gamma m v = 0.28868)
#   In the Kerr rest frame the Schwarzschild BH starts at (x,y) = (-12, -10)
#   moving in +x, i.e. impact parameter b = 10 in y, initial x-distance 12.
#   Orbital angular momentum is along +z (aligned with the Kerr spin).
#
# TwoPunctures can only place punctures on the x-axis, so the whole set-up is
# rotated by -39.8 deg about z: separation d = sqrt(12^2 + 10^2) = 15.6205 now
# lies along x, and the momentum (P, 0, 0) becomes P * (12/d, -10/d, 0).
# A z-rotation leaves the z-spin unchanged. Physically identical, just rotated.
#
# Fly-by estimate: CM-frame J_orb/M^2 = 1.39 at E/M = 1.038. Equal-mass,
# non-spinning capture threshold is J/M^2 ~ 1.1-1.2 (Damour et al. 2014,
# Sperhake et al. 2009), and aligned spin makes capture harder, so this should
# scatter strongly. Lower b towards ~8 for a closer miss (risk of capture),
# raise it for a gentler deflection. Re-derive momentum_minus if you change it.
#
# Resolution: coarse dx = 2, 7 levels, finest dx = 1/32 = m/16. That is low-res
# but not garbage (~6-8 cells across each horizon radius). Domain is
# [-64,64]^2 x [0,64] using z-reflection (equatorial symmetry holds because the
# spin is along z and all motion is in the xy plane).

amr.verbose = 1

# Output

two_punctures.calculate_target_masses = false

# two_punctures.target_mass_minus = 0.5      # Schwarzschild

# Kerr at the grid centre, Schwarzschild at distance d along -x
two_punctures.offset_plus  = 0.0

two_punctures.momentum_plus  = 0.0 0.0 0.0
two_punctures.momentum_minus = 0.221766381286372 -0.184805317738643 0.0

# S = chi m^2 = 0.7 * 0.25
two_punctures.spin_plus  = 0.0 0.0 0.175

# Spectral resolution: modest for a fast local solve (production uses 64 64 48

two_punctures.num_points_A   = 40

# Grid

# Regrid intervals in level steps for levels 0..max_level-1. Levels 2-5 all
# regrid every 0.125 M; at v = 0.5 the BH moves 0.0625 M (2 finest cells)
# between regrids.
amr.regrid_int = 2 2 1 2 4 8

# Puncture tagging: finest level covers r < finest_level_factor * m around each
